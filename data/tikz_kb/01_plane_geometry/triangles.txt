\begin{tikzpicture}[scale=1.5, thick]

=== TEMPLATE: Tam giác vuông ABC tại A ===
TAGS: 2D, tam giác vuông, right triangle
KEYWORDS: tam giác vuông, vuông tại A, right triangle, perpendicular
NOTES: Đỉnh vuông góc A đặt ở góc dưới-trái. Dùng pic right angle (xoay theo
       cạnh) thay vì rectangle (không xoay được).

\begin{tikzpicture}[scale=1.5, thick]
  \coordinate[label=below left:$A$]  (A) at (0,0);
  \coordinate[label=below right:$B$] (B) at (3,0);
  \coordinate[label=above left:$C$]  (C) at (0,2);
  \draw (A)--(B)--(C)--cycle;
  % KÝ HIỆU GÓC VUÔNG CHUẨN — pic right angle (tự xoay theo hai cạnh)
  \draw pic [draw, angle radius=4mm, "$\cdot$" anchor=north east, "" pos=0.5]
    {right angle = B--A--C};
  % Nhãn cạnh
  \node at (1.5,0)   [below]  {$c$};
  \node at (0,1)     [left]   {$b$};
  \node at (1.55,1)  [above right] {$a$};
\end{tikzpicture}


=== TEMPLATE: Tam giác cân ABC tại A ===

\begin{tikzpicture}[scale=1.5, thick,

    mark=at position 0.5 with {\draw (-2pt,-2pt)--(2pt,2pt);}}}]

  \coordinate[label=below left:$B$] (B) at (0,0);
  \coordinate[label=below right:$C$](C) at (4,0);

  \draw[tick] (A)--(C);
  \draw       (B)--(C);
  % Đường cao AH = trung tuyến (vì cân tại A)
  \coordinate[label=below:$H$] (H) at (2,0);
  \draw[dashed] (A)--(H);
  \draw pic [draw, angle radius=3mm] {right angle = A--H--C};
\end{tikzpicture}

=== TEMPLATE: Tam giác có trọng tâm G ===
TAGS: 2D, tam giác, trọng tâm, centroid
KEYWORDS: trọng tâm, centroid, G, AG = 2/3 AM, 3 trung tuyến đồng quy
NOTES: G = giao 3 đường trung tuyến; G chia mỗi trung tuyến theo tỉ số
       2:1 (AG = 2·GM). Alpha 24 fix: dùng tam giác đều cạnh 4,
       G = (0, 2/√3), 3 trung điểm chính xác.

\begin{tikzpicture}[scale=1, thick,
  tick/.style={postaction={decorate}, decoration={markings,
    mark=at position 0.5 with {\draw (-2pt,-2pt)--(2pt,2pt);}}}]
  \coordinate[label=above:$A$]       (A) at ( 0, 3.464);
  \coordinate[label=below left:$B$]  (B) at (-2, 0);
  \coordinate[label=below right:$C$] (C) at ( 2, 0);
  % Trung điểm 3 cạnh (toán đúng):
  \coordinate[label=below:$M$]       (M) at ( 0, 0);        % trung BC
  \coordinate[label=above right:$N$] (N) at ( 1, 1.732);    % trung CA
  \coordinate[label=above left:$P$]  (P) at (-1, 1.732);    % trung AB
  % Trọng tâm G = (0, 2/√3)
  \coordinate (G) at (0, 1.155);
  \draw (A)--(B)--(C)--cycle;
  \draw[dashed] (A)--(M);
  \draw[dashed] (B)--(N);
  \draw[dashed] (C)--(P);
  \fill (G) circle (1.5pt) node[above right=1pt] {$G$};
  \draw[tick] (B)--(M); \draw[tick] (M)--(C);
\end{tikzpicture}
