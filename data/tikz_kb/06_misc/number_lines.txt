=== TEMPLATE: Khoảng nghiệm trên trục số (đoạn đóng [a,b]) ===
TAGS: misc, trục số, đoạn đóng
KEYWORDS: x ∈ [a, b], đoạn đóng, nghiệm bao gồm
NOTES: Hai chấm đặc ● = bao gồm; tô đậm khoảng giữa.

\begin{tikzpicture}[scale=1]
  \draw[->, thick] (-3,0)--(5,0) node[right] {$x$};
  \foreach \x in {-2,-1,0,1,2,3,4} {
    \draw (\x,0.08)--(\x,-0.08);
    \node[below] at (\x,-0.1) {$\x$};
  }
  % Khoảng [-1, 3] đóng — chấm đặc + đường tô đậm
  \draw[blue, very thick] (-1,0.2)--(3,0.2);
  \fill[blue] (-1,0.2) circle (3pt);
  \fill[blue] ( 3,0.2) circle (3pt);
  \node[blue, above] at (1, 0.4) {$x \in [-1; 3]$};
\end{tikzpicture}


=== TEMPLATE: Khoảng nghiệm trên trục số (khoảng mở (a, b)) ===
TAGS: misc, trục số, khoảng mở
KEYWORDS: x ∈ (a, b), khoảng mở, không bao gồm

\begin{tikzpicture}[scale=1]
  \draw[->, thick] (-3,0)--(5,0) node[right] {$x$};

  \draw[blue, fill=white, thick] ( 3,0.2) circle (3pt);
  \node[blue, above] at (1, 0.4) {$x \in (-1; 3)$};
\end{tikzpicture}


=== TEMPLATE: Miền nghiệm hệ bất phương trình (2D) ===
TAGS: misc, miền nghiệm, BPT, region
KEYWORDS: miền nghiệm hệ bất phương trình, hệ BPT 2 ẩn, region

\begin{tikzpicture}[scale=0.9]
  % Trục
  \draw[->, thick] (-1,0)--(6,0) node[right] {$x$};
  \draw[->, thick] (0,-1)--(0,6) node[above] {$y$};
  \foreach \x in {1,2,3,4,5} \draw (\x,0.08)--(\x,-0.08) node[below] {$\x$};
  \foreach \y in {1,2,3,4,5} \draw (0.08,\y)--(-0.08,\y) node[left] {$\y$};
  % Đường biên
  \draw[blue, thick]   (-1, 5)--(5,-1);            % x + y = 4
  \draw[red, thick]    (-1,-1)--(5, 5);            % y = x
  \node[blue, above] at (4.2, -0.3) {$x+y=4$};
  \node[red, right]  at (4, 4.5) {$y = x$};
  % Vùng tô (giả định: x+y ≤ 4, y ≤ x, x ≥ 0, y ≥ 0)
  \fill[orange, opacity=0.3] (0,0)--(4,0)--(2,2)--cycle;
\end{tikzpicture}


=== TEMPLATE: Elip tâm O ===
TAGS: 2D, conic, elip
KEYWORDS: elip, ellipse, x^2/a^2 + y^2/b^2 = 1

\begin{tikzpicture}
\begin{axis}[
  axis lines=middle, xlabel=$x$, ylabel=$y$,
  xmin=-4, xmax=4, ymin=-3, ymax=3,
  width=10cm, height=8cm, samples=200,
]
  \addplot[blue, thick, domain=-3:3]  { sqrt(4*(1 - x^2/9)) };
  \addplot[blue, thick, domain=-3:3]  {-sqrt(4*(1 - x^2/9)) };
  \addlegendentry{$\frac{x^2}{9}+\frac{y^2}{4}=1$}
  % Tiêu điểm F1, F2 (c² = a² - b² = 9-4 = 5)
  \addplot[only marks, mark=*, red] coordinates {(2.236,0) (-2.236,0)};
  \node[red, below] at (axis cs: 2.236, 0) {$F_2$};
  \node[red, below] at (axis cs:-2.236, 0) {$F_1$};
\end{axis}
\end{tikzpicture}
